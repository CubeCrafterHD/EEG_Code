import numpy as np
import matplotlib.pyplot as plt
from scipy.signal import stft
from scipy.interpolate import interp1d
from scipy.signal import butter, filtfilt
from scipy.signal import iirnotch
from scipy.signal import medfilt
from scipy.ndimage import gaussian_filter1d

# PARAMETER
FS = 500  # Samplingrate

BANDS = {
    "Delta": (0.5, 4),
    "Theta": (4, 8),
    "Alpha": (8, 13),
    "Beta":  (13, 30),
    "Gamma": (30, 45)
}

BAND_COLORS = {
    "Delta": "tab:blue",
    "Theta": "tab:orange",
    "Alpha": "tab:green",
    "Beta":  "tab:red",
    "Gamma": "tab:purple"
}

# function_bandpass
def bandpass_filter(signal, fs, low=0.5, high=45, order=4):
    nyq = 0.5 * fs
    low = low / nyq
    high = high / nyq
    b, a = butter(order, [low, high], btype='band')
    return filtfilt(b, a, signal)


#function_notch
def notch_filter(signal, fs, freq=50, Q=30):
    b, a = iirnotch(freq, Q, fs)
    return filtfilt(b, a, signal)




# DATEN LADEN
data = np.loadtxt("Papa\Thomas_5_Traubenzucker.csv", delimiter=",", skiprows=1)
signal = data[:, 1]

# DC-Offset
signal = signal - np.mean(signal)

# bandpass
signal = bandpass_filter(signal, FS)

#notchfilter
signal = notch_filter(signal, FS)

#medfilt
signal = medfilt(signal, kernel_size=5)

# Zeitachse
time = np.arange(len(signal)) / FS

# Artefakte entfernen
z = (signal - np.mean(signal)) / np.std(signal)
artifact_mask = np.abs(z) > 4

signal[artifact_mask] = np.nan

valid = ~np.isnan(signal)
interp = interp1d(time[valid], signal[valid], fill_value="extrapolate")
signal = interp(time)



# ZEIT-FREQUENZ-ANALYSE
f, t_stft, Zxx = stft(signal, fs=FS, nperseg=128, noverlap=64)
power = np.abs(Zxx) ** 2


# BANDPOWER
band_power = {}
for band, (low, high) in BANDS.items():
    idx = np.logical_and(f >= low, f <= high)
    band_power[band] = power[idx, :].mean(axis=0)

# Interpolation
band_power_interp = {}
for band in BANDS:
    f_interp = interp1d(t_stft, band_power[band], kind='linear', fill_value="extrapolate")
    band_power_interp[band] = f_interp(time)
    




# PLOTS

# Rohsignal
plt.figure(figsize=(14,4))
plt.plot(time, signal, linewidth=0.8)
plt.title("EEG-Rohsignal")
plt.xlabel("Zeit (s)")
plt.ylabel("Amplitude")
plt.grid(True)
plt.show()

# Bandpower
fig, axes = plt.subplots(len(BANDS), 1, figsize=(14, 10), sharex=True)
for ax, (band, power_values) in zip(axes, band_power_interp.items()):
    ax.plot(time, power_values, label=f"{band}-Band", color=BAND_COLORS[band])
    ax.set_ylabel("Leistung")
    ax.set_title(f"{band}-Band")
    ax.grid(True)
axes[-1].set_xlabel("Zeit (s)")
plt.tight_layout()
plt.show()



# Konzentrationsindex
concentration_index = band_power_interp["Beta"] / (band_power_interp["Theta"] + band_power_interp["Alpha"])

z = (concentration_index - np.mean(concentration_index)) / np.std(concentration_index)
artifact_mask = np.abs(z) > 3

concentration_index[artifact_mask] = np.nan
valid = ~np.isnan(concentration_index)
interp = interp1d(time[valid], concentration_index[valid], fill_value="extrapolate")
concentration_index = interp(time)

#glätten
concentration_index = gaussian_filter1d(concentration_index, sigma=3)

# Plot
plt.figure(figsize=(14,3))
plt.plot(time, concentration_index, color="tab:red")
plt.title("Konzentrations-Index (Beta / (Theta + Alpha))")
plt.xlabel("Zeit (s)")
plt.ylabel("Index")
plt.grid(True)


mean_index = np.mean(concentration_index)
print("Durchschnittlicher Konzentrations-Index:", mean_index)

plt.text(
    x=time[-1],                  
    y=max(concentration_index)*0.9,  
    s=f"Durchschnitt: {mean_index:.2f}",
    color="black",
    fontsize=12,
    ha="right",
    va="top",
    bbox=dict(facecolor='white', alpha=0.7, edgecolor='gray')
)

plt.show()
